// Runtime assertions for v0.4 typed-enum channel parameters.

package enumparamsv1

import (
	"context"
	"slices"
	"strings"
	"testing"
)

type recPub struct {
	exchange, routingKey string
	body                 []byte
}

func (r *recPub) Publish(_ context.Context, exchange, routingKey string, body []byte, _ PublishProperties) error {
	r.exchange = exchange
	r.routingKey = routingKey
	r.body = body
	return nil
}

type recSub struct {
	queue       string
	bindingKeys []string
	handler     func(ctx context.Context, routingKey string, body []byte) error
}

func (r *recSub) Subscribe(_ context.Context, queueName string, bindingKeys []string,
	handler func(ctx context.Context, routingKey string, body []byte) error) error {
	r.queue = queueName
	r.bindingKeys = bindingKeys
	r.handler = handler
	return nil
}

// TestSendJob_AcceptsTypedEnumOnly proves the SendJob jobType arg is
// the JobType type (not string). Passing a constant works; passing a
// string literal would be a compile error in this same package.
func TestSendJob_AcceptsTypedEnumOnly(t *testing.T) {
	tr := &recPub{}
	if err := NewPublisher(tr).SendJob(context.Background(), "acme", JobTypeDeploy, JobMessage{ID: "j-1"}); err != nil {
		t.Fatalf("Send: %v", err)
	}
	if tr.routingKey != "acme.deploy" {
		t.Errorf("routingKey = %q, want acme.deploy", tr.routingKey)
	}
}

// TestSubscribeJobResult_AcceptsTypedEnum ensures the queue-name
// builder also takes the typed enum and serialises it correctly.
func TestSubscribeJobResult_AcceptsTypedEnum(t *testing.T) {
	tr := &recSub{}
	type handlerFn func(ctx context.Context, msg JobResult) error
	var h handlerFn = func(_ context.Context, _ JobResult) error { return nil }
	if err := NewSubscriber(tr).SubscribeJobResult(context.Background(), JobTypeRollback, jrHandlerAdapter(h)); err != nil {
		t.Fatalf("Subscribe: %v", err)
	}
	if tr.queue != "rollback.results" {
		t.Errorf("queue = %q, want rollback.results", tr.queue)
	}
	// The binding-key expression applies the same string(enum) cast.
	if len(tr.bindingKeys) != 1 || tr.bindingKeys[0] != "rollback.results" {
		t.Errorf("bindingKeys = %v, want [rollback.results]", tr.bindingKeys)
	}
}

type jrHandlerAdapter func(ctx context.Context, msg JobResult) error

func (a jrHandlerAdapter) HandleJobResult(ctx context.Context, msg JobResult) error {
	return a(ctx, msg)
}

// TestJobTypeConstants_HaveExpectedValues guards against accidental
// drift in const naming or value mapping. The generated names follow
// the <TypeName><Pascalize(value)> convention.
func TestJobTypeConstants_HaveExpectedValues(t *testing.T) {
	cases := []struct {
		c    JobType
		want string
	}{
		{JobTypeBuild, "build"},
		{JobTypeDeploy, "deploy"},
		{JobTypeRollback, "rollback"},
	}
	for _, tc := range cases {
		if string(tc.c) != tc.want {
			t.Errorf("%v != %q", tc.c, tc.want)
		}
	}
}

// TestTenantStaysString proves that parameters without an enum or a
// pattern keep the default `string` Go type.
func TestTenantStaysString(t *testing.T) {
	// The arg type appears in the function signature; a compile-time
	// check is enough — if SendJob's tenant arg had been typed, this
	// call would fail to compile (we're passing a plain string literal).
	tr := &recPub{}
	_ = NewPublisher(tr).SendJob(context.Background(), "acme-tenant", JobTypeBuild, JobMessage{ID: "x"})
}

// TestJobTypeValues_InSpecOrder checks the generated values function,
// and that it returns a new slice each time.
func TestJobTypeValues_InSpecOrder(t *testing.T) {
	got := JobTypeValues()
	if want := []JobType{JobTypeBuild, JobTypeDeploy, JobTypeRollback}; !slices.Equal(got, want) {
		t.Errorf("JobTypeValues() = %q, want %q", got, want)
	}
	got[0] = "changed"
	if JobTypeValues()[0] != JobTypeBuild {
		t.Error("JobTypeValues returns a slice its caller can change")
	}
}

func TestJobType_Valid(t *testing.T) {
	for _, v := range JobTypeValues() {
		if !v.Valid() {
			t.Errorf("%q.Valid() = false", v)
		}
	}
	for _, v := range []JobType{"", "bogus", "Build"} {
		if v.Valid() {
			t.Errorf("%q.Valid() = true", v)
		}
	}
}

// TestSendJob_RejectsValueOutsideTheEnum: a JobType made by conversion
// compiles, so SendJob checks it, and publishes nothing.
func TestSendJob_RejectsValueOutsideTheEnum(t *testing.T) {
	tr := &recPub{}
	err := NewPublisher(tr).SendJob(context.Background(), "acme", JobType("bogus"), JobMessage{ID: "j-1"})
	if err == nil || !strings.Contains(err.Error(), `parameter jobType: "bogus" is not a valid JobType`) {
		t.Fatalf("SendJob error = %v, want one naming the parameter", err)
	}
	if tr.routingKey != "" {
		t.Errorf("published to %q", tr.routingKey)
	}
}

// TestSubscribeJobResult_RejectsValueOutsideTheEnum: Subscribe checks
// the value too, before it binds anything.
func TestSubscribeJobResult_RejectsValueOutsideTheEnum(t *testing.T) {
	tr := &recSub{}
	h := jrHandlerAdapter(func(context.Context, JobResult) error { return nil })
	err := NewSubscriber(tr).SubscribeJobResult(context.Background(), JobType("bogus"), h)
	if err == nil || !strings.Contains(err.Error(), `parameter jobType: "bogus" is not a valid JobType`) {
		t.Fatalf("SubscribeJobResult error = %v, want one naming the parameter", err)
	}
	if tr.queue != "" {
		t.Errorf("subscribed to %q", tr.queue)
	}
}
